# Compile the same contract against each supported implementation boundary and
# both runtime-selected and compile-time priority counts.
foreach(mode IN ITEMS header standalone shared)
    foreach(priorities IN ITEMS runtime fixed)
        set(name tmc-notification-${mode}-${priorities})
        add_executable(${name} manual_executor_notification.cpp)
        target_link_libraries(${name} PRIVATE TooManyCooks::TooManyCooks)
        if(priorities STREQUAL fixed)
            target_compile_definitions(${name} PRIVATE TMC_PRIORITY_COUNT=2)
        endif()

        if(NOT mode STREQUAL header)
            if(mode STREQUAL shared)
                add_library(${name}-implementation SHARED implementation.cpp)
                target_compile_definitions(${name}-implementation PUBLIC TMC_SHARED_LIBRARY)
            else()
                add_library(${name}-implementation STATIC implementation.cpp)
                target_compile_definitions(${name}-implementation PUBLIC TMC_STANDALONE_COMPILATION)
            endif()
            target_link_libraries(${name}-implementation PUBLIC TooManyCooks::TooManyCooks)
            target_compile_definitions(${name}-implementation PRIVATE TMC_IMPL)
            if(priorities STREQUAL fixed)
                target_compile_definitions(${name}-implementation PUBLIC TMC_PRIORITY_COUNT=2)
            endif()
            target_link_libraries(${name} PRIVATE ${name}-implementation)
        endif()

        add_test(NAME ${name} COMMAND ${name})
        set_tests_properties(${name} PROPERTIES TIMEOUT 20)
    endforeach()
endforeach()
